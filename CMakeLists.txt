cmake_minimum_required(VERSION 3.18)
project(qdjj_solver LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
option(QDJJ_COVERAGE "Instrument first-party native code for gcov/llvm-cov" OFF)
option(QDJJ_SANITIZERS "Instrument the extension with address/undefined sanitizers" OFF)
if(QDJJ_COVERAGE AND QDJJ_SANITIZERS)
  message(FATAL_ERROR "Use separate build directories for coverage and sanitizers")
endif()
if(QDJJ_COVERAGE OR QDJJ_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR MSVC)
    message(FATAL_ERROR "Native instrumentation requires GCC or Clang")
  endif()
  # Avoid pybind11's release LTO and stripping in instrumented builds.
  pybind11_add_module(_core NO_EXTRAS cpp/core.cpp)
  set_target_properties(_core PROPERTIES INTERPROCEDURAL_OPTIMIZATION OFF)
else()
  pybind11_add_module(_core cpp/core.cpp)
endif()
target_compile_features(_core PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(_core PRIVATE /W4)
else()
  target_compile_options(_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(QDJJ_COVERAGE)
  target_compile_options(_core PRIVATE -O0 -g --coverage)
  target_link_options(_core PRIVATE --coverage)
elseif(QDJJ_SANITIZERS)
  target_compile_options(_core PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=all)
  target_link_options(_core PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS _core LIBRARY DESTINATION qdjj_solver/qp_solver RUNTIME DESTINATION qdjj_solver/qp_solver)
